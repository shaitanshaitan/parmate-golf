rules_version = '2';

// ─────────────────────────────────────────────────────────────────────────
// Parmate Player App + Event Manager — Firestore rules  (whs-teamhcp-trg-26-app)
// Added: events/{eventId} + flights/{flightId} — organiser-only (Event Manager).
// Added (v26, 19 Oct 2026): m1live/{matchId} — live 1vs1 between two buddies (Player App).
// Everything else unchanged from your tightened ruleset.
// Test in the Rules Playground before Publish.
// ─────────────────────────────────────────────────────────────────────────

service cloud.firestore {
  match /databases/{database}/documents {

    // Admin granted via custom claim set from the Event Manager portal.
    function isAdmin() { return request.auth != null && request.auth.token.admin == true; }

    // ── Messaging helpers ──────────────────────────────────────────────────
    function other(parts) { return parts[0] == request.auth.uid ? parts[1] : parts[0]; }
    function notBlockedBy(rcpt) {
      return !exists(/databases/$(database)/documents/users/$(rcpt)/blocks/$(request.auth.uid));
    }
    function isBuddyOf(rcpt) {
      return exists(/databases/$(database)/documents/users/$(request.auth.uid)/buddies/$(rcpt));
    }
    function threadParts(tid) {
      return get(/databases/$(database)/documents/threads/$(tid)).data.participants;
    }

    // Owner owns everything under their own tree.
    match /users/{userId}/{document=**} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }

    match /users/{userId}/teams/{teamId} {
      allow create: if request.auth != null && request.resource.data.buddyUid == request.auth.uid;
      allow read:   if request.auth != null;
    }

    match /users/{userId}/buddies/{buddyId} {
      allow read: if request.auth != null;
      allow create, delete: if request.auth != null && buddyId == request.auth.uid;
    }
    match /users/{userId}/rounds/{roundId}  { allow read: if request.auth != null; }

    match /users/{userId}/buddyRequests/{reqId} {
      allow create: if request.auth != null
        && !exists(/databases/$(database)/documents/users/$(userId)/blocks/$(request.auth.uid));
    }

    match /users/{userId}/notifications/{notifId} {
      allow read, update: if request.auth != null && request.auth.uid == userId;
      allow create:       if request.auth != null;
    }

    match /public_profiles/{userId} {
      allow read:  if request.auth != null;
      allow write: if request.auth != null && request.auth.uid == userId;
    }

    match /matches/{matchId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null &&
        request.auth.uid == request.resource.data.teamA.uid;
      allow update: if request.auth != null && (
        request.auth.uid == resource.data.teamA.uid ||
        (resource.data.status == 'waiting' &&
         resource.data.teamB.uid == null &&
         request.auth.uid != resource.data.teamA.uid) ||
        request.auth.uid == resource.data.teamB.uid
      );
      allow delete: if request.auth != null &&
        (request.auth.uid == resource.data.teamA.uid ||
         request.auth.uid == resource.data.teamB.uid);
    }

    // ── 1vs1 Live Match between two buddies (Player App v26) ────────────────
    // Only the two players can see it. The host can only start one with a buddy.
    // During the match both players may enter scores, join, decline and save;
    // the setup (course, players, handicaps, strokes) can't be changed after START.
    match /m1live/{matchId} {
      allow read: if request.auth != null
        && request.auth.uid in resource.data.players;
      allow create: if request.auth != null
        && request.resource.data.uidA == request.auth.uid
        && request.resource.data.uidB is string
        && request.resource.data.uidB != request.auth.uid
        && request.resource.data.players == [request.auth.uid, request.resource.data.uidB]
        && isBuddyOf(request.resource.data.uidB);
      allow update: if request.auth != null
        && request.auth.uid in resource.data.players
        && request.resource.data.diff(resource.data).affectedKeys()
             .hasOnly(['h', 'extraCount', 'halved', 'status', 'joinedB', 'declinedB',
                       'savedA', 'savedB', 'updatedAt', 'lastBy']);
      allow delete: if request.auth != null && (
        request.auth.uid == resource.data.uidA ||
        (request.auth.uid == resource.data.uidB && resource.data.status == 'complete'));
    }

    match /courses_shared/{courseId} {
      allow read:   if request.auth != null;
      allow create: if request.auth != null;
      allow update: if request.auth != null;
      allow delete: if false;
    }

    // ── Event Manager — event documents (teams, tees, format, code, status) ─
    // Strictly organiser-only. Nobody outside the organiser's own account can
    // read or write event details, teams, or player HCP/gender data.
    match /events/{eventId} {
      allow read:                if request.auth != null && resource.data.organiserId == request.auth.uid;
      allow create:              if request.auth != null && request.resource.data.organiserId == request.auth.uid;
      allow update, delete:      if request.auth != null && resource.data.organiserId == request.auth.uid;
    }

    // ── Event Manager — flight documents (Strokeplay/Matchplay 2-team groupings) ─
    // Organiser-only for now. Each flight carries organiserId + eventId + flightCode.
    // NOTE: when the Player App "join via flight code" feature is built, this will
    // gain a narrow rule letting a joined player update ONLY their own team's
    // scores[] — deliberately not added yet to keep the surface locked down.
    match /flights/{flightId} {
      allow read:                if request.auth != null && resource.data.organiserId == request.auth.uid;
      allow create:              if request.auth != null && request.resource.data.organiserId == request.auth.uid;
      allow update, delete:      if request.auth != null && resource.data.organiserId == request.auth.uid;
    }

    match /faq_articles/{id} {
      allow read:  if request.auth != null;
      allow write: if isAdmin();
    }

    match /deletion_requests/{id} {
      allow create: if request.auth != null;
      allow read, update, delete: if isAdmin();
    }

    match /threads/{threadId} {
      allow read: if request.auth != null
        && request.auth.uid in resource.data.participants;
      allow create: if request.auth != null
        && request.resource.data.participants.size() == 2
        && request.auth.uid in request.resource.data.participants
        && request.resource.data.participants[0] < request.resource.data.participants[1]
        && threadId == request.resource.data.participants[0] + '_' + request.resource.data.participants[1]
        && isBuddyOf(other(request.resource.data.participants))
        && notBlockedBy(other(request.resource.data.participants));
      allow update: if request.auth != null
        && request.auth.uid in resource.data.participants;
      allow delete: if false;

      match /messages/{msgId} {
        allow read: if request.auth != null
          && request.auth.uid in threadParts(threadId);
        allow create: if request.auth != null
          && request.resource.data.from == request.auth.uid
          && request.auth.uid in threadParts(threadId)
          && request.resource.data.text is string
          && request.resource.data.text.size() >= 1
          && request.resource.data.text.size() <= 280
          && notBlockedBy(other(threadParts(threadId)));
        allow update, delete: if false;
      }
    }

    match /config/{doc} {
      allow read:  if request.auth != null;
      allow write: if isAdmin();
    }

  }
}
